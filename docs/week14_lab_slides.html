# database.py # Author: [Your Name] import sqlite3 def query_all(db_file, sql, values=()): """Run a SELECT and return every row as a dictionary.""" conn = sqlite3.connect(db_file) conn.row_factory = sqlite3.Row rows = conn.execute(sql, values).fetchall() conn.close() return [dict(row) for row in rows] def query_one(db_file, sql, values=()): """Run a SELECT and return the first row as a dictionary, or None.""" rows = query_all(db_file, sql, values) if rows == []: return None return rows[0] def change(db_file, sql, values=()): """Run an INSERT or UPDATE, save it, and return the ID of the last row inserted.""" conn = sqlite3.connect(db_file) cursor = conn.execute(sql, values) conn.commit() row_id = cursor.lastrowid conn.close() return row_id def create_tables(db_file): """Create the tables and the sample rows, unless they are already there.""" with open("schema.sql") as f: script = f.read() conn = sqlite3.connect(db_file) conn.executescript(script) conn.commit() conn.close()
